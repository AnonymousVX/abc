PRACTICAL 7: TRANSFER LEARNING WITH A PRETRAINED CNN
Framework: PyTorch | Notebook: Jupyter | Dataset: Cats vs Dogs
Model: ResNet50

IMPORTANT:
1. Extract the Cats vs Dogs dataset before running this code.
2. The dataset folder must contain two class folders, for example:
   cats_vs_dogs/
       cats_set/
           cat1.jpg
           cat2.jpg
       dogs_set/
           dog1.jpg
           dog2.jpg
3. Change DATASET_PATH in Cell 2 to your actual extracted folder.
4. ResNet50 pretrained weights must already be cached for fully offline use.
   If they are not cached, load the weights once while internet is available.
5. This example splits the local folder into training and testing subsets.

------------------------------------------------------------
CELL 1: IMPORT LIBRARIES
------------------------------------------------------------

import os
import torch
import torch.nn as nn
import torch.optim as optim
import torchvision.transforms as transforms
import matplotlib.pyplot as plt
import numpy as np

from torch.utils.data import DataLoader, Subset
from torchvision.datasets import ImageFolder
from torchvision.models import resnet50, ResNet50_Weights
from PIL import Image


------------------------------------------------------------
CELL 2: SET DEVICE AND DATASET PATH
------------------------------------------------------------

device = torch.device("cuda" if torch.cuda.is_available() else "cpu")
print("Using device:", device)

# CHANGE THIS PATH ONLY
DATASET_PATH = r"C:\Users\Vinay\Desktop\Dataset_DL\cats_vs_dogs"

if not os.path.isdir(DATASET_PATH):
    raise FileNotFoundError("Dataset folder not found. Check DATASET_PATH.")

print("Dataset folders:", os.listdir(DATASET_PATH))


------------------------------------------------------------
CELL 3: IMAGE TRANSFORMATIONS
------------------------------------------------------------

train_transform = transforms.Compose([
    transforms.Resize((224, 224)),
    transforms.RandomHorizontalFlip(),
    transforms.RandomRotation(10),
    transforms.ToTensor(),
    transforms.Normalize(
        [0.485, 0.456, 0.406],
        [0.229, 0.224, 0.225]
    )
])

test_transform = transforms.Compose([
    transforms.Resize((224, 224)),
    transforms.ToTensor(),
    transforms.Normalize(
        [0.485, 0.456, 0.406],
        [0.229, 0.224, 0.225]
    )
])


------------------------------------------------------------
CELL 4: LOAD DATASET AND CREATE TRAIN/TEST SPLIT
------------------------------------------------------------

# Load the same files twice so training and testing can use
# different transformations.
train_data = ImageFolder(
    root=DATASET_PATH,
    transform=train_transform
)

test_data = ImageFolder(
    root=DATASET_PATH,
    transform=test_transform
)

print("Classes:", train_data.classes)
print("Total images:", len(train_data))

if len(train_data) < 2:
    raise ValueError("Dataset must contain at least two images.")

# Random 80% training and 20% testing split
np.random.seed(42)
indices = np.random.permutation(len(train_data))

split = int(0.8 * len(indices))
train_indices = indices[:split]
test_indices = indices[split:]

# Optional smaller subset for faster practical execution.
# Change these values or remove the slicing to use all images.
train_indices = train_indices[:2000]
test_indices = test_indices[:500]

train_dataset = Subset(train_data, train_indices)
test_dataset = Subset(test_data, test_indices)

train_loader = DataLoader(
    train_dataset,
    batch_size=8,
    shuffle=True,
    num_workers=0
)

test_loader = DataLoader(
    test_dataset,
    batch_size=8,
    shuffle=False,
    num_workers=0
)

print("Training images:", len(train_dataset))
print("Testing images:", len(test_dataset))


------------------------------------------------------------
CELL 5: LOAD PRETRAINED RESNET50
------------------------------------------------------------

weights = ResNet50_Weights.DEFAULT
model = resnet50(weights=weights)

print("Pretrained ResNet50 loaded!")


------------------------------------------------------------
CELL 6: FREEZE LAYERS AND REPLACE FINAL CLASSIFIER
------------------------------------------------------------

# Freeze all pretrained layers
for param in model.parameters():
    param.requires_grad = False

# Replace final classification layer for 2 classes
num_features = model.fc.in_features

model.fc = nn.Sequential(
    nn.Dropout(0.5),
    nn.Linear(num_features, 2)
)

model = model.to(device)

print("Pretrained layers frozen!")
print("New classification layer added!")


------------------------------------------------------------
CELL 7: LOSS FUNCTION AND OPTIMIZER
------------------------------------------------------------

criterion = nn.CrossEntropyLoss()

optimizer = optim.Adam(
    model.fc.parameters(),
    lr=0.001
)


------------------------------------------------------------
CELL 8: TRAINING AND EVALUATION FUNCTION
------------------------------------------------------------

def train_model(model, train_loader, test_loader,
                criterion, optimizer, epochs):

    train_losses = []
    test_accuracies = []

    for epoch in range(epochs):

        model.train()
        running_loss = 0

        for images, labels in train_loader:

            images = images.to(device)
            labels = labels.to(device)

            outputs = model(images)
            loss = criterion(outputs, labels)

            optimizer.zero_grad()
            loss.backward()
            optimizer.step()

            running_loss += loss.item()

        avg_loss = running_loss / len(train_loader)
        train_losses.append(avg_loss)

        # Evaluate model
        model.eval()
        correct = 0
        total = 0

        with torch.no_grad():
            for images, labels in test_loader:

                images = images.to(device)
                labels = labels.to(device)

                outputs = model(images)
                _, predicted = torch.max(outputs, 1)

                total += labels.size(0)
                correct += (predicted == labels).sum().item()

        accuracy = 100 * correct / total
        test_accuracies.append(accuracy)

        print(
            f"Epoch [{epoch+1}/{epochs}] "
            f"Loss: {avg_loss:.4f} "
            f"Accuracy: {accuracy:.2f}%"
        )

    return train_losses, test_accuracies


------------------------------------------------------------
CELL 9: TRAIN FROZEN RESNET50
------------------------------------------------------------

print("Training Frozen ResNet50")

frozen_loss, frozen_accuracy = train_model(
    model,
    train_loader,
    test_loader,
    criterion,
    optimizer,
    epochs=5
)


------------------------------------------------------------
CELL 10: UNFREEZE LAST RESNET50 BLOCK
------------------------------------------------------------

for param in model.layer4.parameters():
    param.requires_grad = True

print("Last ResNet50 block unfrozen!")


------------------------------------------------------------
CELL 11: FINE-TUNE RESNET50
------------------------------------------------------------

optimizer = optim.Adam(
    filter(lambda p: p.requires_grad, model.parameters()),
    lr=0.00001
)

print("Fine-tuning ResNet50")

fine_loss, fine_accuracy = train_model(
    model,
    train_loader,
    test_loader,
    criterion,
    optimizer,
    epochs=5
)


------------------------------------------------------------
CELL 12: COMPARE PERFORMANCE
------------------------------------------------------------

print("FINAL RESULTS")
print("-------------------------")

print(
    "Frozen ResNet50 Accuracy:",
    f"{frozen_accuracy[-1]:.2f}%"
)

print(
    "Fine-tuned ResNet50 Accuracy:",
    f"{fine_accuracy[-1]:.2f}%"
)

improvement = fine_accuracy[-1] - frozen_accuracy[-1]
print("Accuracy Change:", f"{improvement:+.2f}%")


------------------------------------------------------------
CELL 13: PLOT ACCURACY COMPARISON
------------------------------------------------------------

plt.figure(figsize=(8, 5))

plt.plot(frozen_accuracy, label="Frozen ResNet50")
plt.plot(fine_accuracy, label="Fine-tuned ResNet50")

plt.xlabel("Epochs")
plt.ylabel("Accuracy (%)")
plt.title("Frozen vs Fine-tuned Accuracy")
plt.legend()
plt.grid()
plt.show()


------------------------------------------------------------
CELL 14: PLOT LOSS COMPARISON
------------------------------------------------------------

plt.figure(figsize=(8, 5))

plt.plot(frozen_loss, label="Frozen ResNet50")
plt.plot(fine_loss, label="Fine-tuned ResNet50")

plt.xlabel("Epochs")
plt.ylabel("Training Loss")
plt.title("Frozen vs Fine-tuned Loss")
plt.legend()

------------------------------------------------------------
CELL 15: PREDICT A NEW IMAGE
------------------------------------------------------------

image_path = input("Enter image path: ")

image = Image.open(image_path).convert("RGB")
input_image = test_transform(image).unsqueeze(0).to(device)

model.eval()

with torch.no_grad():
    output = model(input_image)
    _, prediction = torch.max(output, 1)

classes = train_data.classes

print("Predicted Class:", classes[prediction.item()])

plt.imshow(image)

------------------------------------------------------------
VIVA / IMPORTANT POINTS
------------------------------------------------------------

1. Transfer learning:
   Reusing a model trained on a large dataset for a new task.

2. ResNet50:
   A 50-layer convolutional neural network.

3. Freezing:
   Setting requires_grad=False so pretrained weights are not updated.

4. Fine-tuning:
   Unfreezing selected pretrained layers and training them on the new dataset.

5. ImageFolder:
   Loads images from subfolders. Each subfolder name becomes a class.

6. CrossEntropyLoss:
   Loss function used for multi-class classification.

7. Adam:
   Optimizer used to update trainable model weights.

8. ResNet50 input:
   RGB images resized to 224 x 224 pixels.

NOTE:
- The example uses up to 2,000 training and 500 testing images for quicker execution.
- The test subset is sampled from the same folder, so use a separate test folder if your dataset provides one for a more reliable evaluation.
- If pretrained weights are not cached locally, the first ResNet50 load requires internet access.
